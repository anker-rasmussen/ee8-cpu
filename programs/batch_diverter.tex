% ── Title ────────────────────────────────────────────────────────────
\begin{center}
  {\LARGE\bfseries EE8 --- Batch Diverter}\\[6pt]
  {\large 5 items to Gate~A, 3 items to Gate~B, repeat}
\end{center}
\vspace{1em}

\noindent
The batch diverter sends 5~items to Gate~A, then 3~items to Gate~B,
alternating forever. It uses the full v4 platform: I/O ports
(\texttt{IN}/\texttt{OUT}), the \texttt{ITEM\_PULSE} sensor latch,
and the motor safety interlock. All 16~instruction slots are used.

% ═════════════════════════════════════════════════════════════════════
\section{I/O Port Bits}
% ═════════════════════════════════════════════════════════════════════

\begin{center}
\begin{tabular}{lll}
  \toprule
  \textbf{Port} & \textbf{Bit} & \textbf{Signal} \\
  \midrule
  Input  & 0 & \texttt{ITEM\_PULSE} --- set by sensor, cleared on \texttt{IN} \\
  Output & 0 & \texttt{GATE\_SEL} --- 0\,=\,Gate~A, 1\,=\,Gate~B \\
  Output & 1 & \texttt{MOTOR\_EN} --- motor enable request \\
  \bottomrule
\end{tabular}
\end{center}

\noindent
\hex{02}\,=\,\texttt{0b00000010}: motor on, Gate~A.\quad
\hex{03}\,=\,\texttt{0b00000011}: motor on, Gate~B.

% ═════════════════════════════════════════════════════════════════════
\section{Assembly Program}
% ═════════════════════════════════════════════════════════════════════

\begin{lstlisting}[style=asmstyle]
; ── Batch Diverter: A=5, B=3, alternating ────────────
;
; Setup
0:  LDI  R0, 5          ; Batch count A
1:  LDI  R2, 0x02       ; MOTOR_EN=1, GATE_SEL=0 (Gate A)

; Counting loop (shared for both gates)
2:  OUT  R2              ; Drive output port
3:  IN   R1              ; Read input (clears ITEM_PULSE)
4:  LDI  R3, 0x01       ; ITEM_PULSE mask
5:  AND  R1, R3, R1     ; Isolate bit 0
6:  JZ   R1, 3          ; No item? Keep polling
7:  LDI  R1, 1          ; Decrement constant
8:  SUB  R0, R1, R0     ; R0--
9:  JNZ  R0, 3          ; More items? Next item

; Batch complete: toggle gate, reload count
10: LDI  R1, 0x01       ; GATE_SEL toggle mask
11: XOR  R2, R1, R2     ; Flip bit 0: 0x02 <-> 0x03
12: AND  R2, R1, R3     ; R3 = new GATE_SEL (0 or 1)
13: LDI  R0, 3          ; Load count B
14: JNZ  R3, 2          ; Gate B? Start counting
15: LDI  R0, 5          ; Gate A: load count A
                         ; PC wraps 15->0->1->2
\end{lstlisting}

\begin{notebox}
After Gate~A completes, \texttt{XOR} flips \reg{2} from \hex{02} to
\hex{03}. \texttt{AND} with \hex{01} extracts the new \texttt{GATE\_SEL}
into \reg{3}. If \reg{3}\,=\,1 (Gate~B), \texttt{JNZ} jumps to address~2.
If \reg{3}\,=\,0 (Gate~A), execution falls through to address~15 which
loads count~A, then the PC wraps to address~0 (redundant \texttt{LDI R0,5})
and address~1 (restores \reg{2}\,=\,\hex{02}).
\end{notebox}
